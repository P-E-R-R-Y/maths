165 // Minimum Spanning Tree via Kruskal. Returns a new graph sharing the same
166 // node indexing as `*this`, containing only the MST edges. The original
167 // is untouched, so caller can compute `difference(*this, mst)` for loopback.
168 Graph mst() const {
169 Graph result;
170 result.nodes_ = nodes_;
171 if (nodes_.empty() || edges_.empty()) return result;
172
173 std::vector<Link> sorted = edges_;
174 std::sort(sorted.begin(), sorted.end(),
175 [](const Link& a, const Link& b) { return a.weight < b.weight; });
176
177 std::vector<int> parent(nodes_.size());
178 std::vector<int> rank_(nodes_.size(), 0);
179 std::iota(parent.begin(), parent.end(), 0);
180
181 auto find = [&](int x) {
182 while (x != parent[x]) {
183 parent[x] = parent[parent[x]]; // path halving
184 x = parent[x];
185 }
186 return x;
187 };
188 auto unite = [&](int a, int b) -> bool {
189 int ra = find(a), rb = find(b);
190 if (ra == rb) return false;
191 if (rank_[ra] < rank_[rb]) std::swap(ra, rb);
192 parent[rb] = ra;
193 if (rank_[ra] == rank_[rb]) ++rank_[ra];
194 return true;
195 };
196
197 result.reserveEdges(nodes_.size() - 1);
198 for (const auto& e : sorted) {
199 if (unite(e.from, e.to)) result.edges_.push_back(e);
200 }
201 return result;
202 }

204 // Edges in *this that are NOT in `other`. Useful for TinyKeep-style
205 // "Delaunay minus MST" to recover loopback candidates.
206 // Both graphs must share the same node indexing.
207 Graph difference(const Graph& other) const {
208 Graph result;
209 result.nodes_ = nodes_;
210
211 std::set<std::pair<int,int>> keys;
212 for (const auto& e : other.edges_) keys.insert(edgeKey(e));
213
214 for (const auto& e : edges_) {
215 if (!keys.count(edgeKey(e))) result.edges_.push_back(e);
216 }
217 return result;
218 }

228 // Connected components as lists of node indices. Honors graph kind: in
229 // directed mode this is *weakly* connected components.
230 std::vector<std::vector<int>> connectedComponents() const {
231 std::vector<int> parent(nodes_.size());
232 std::iota(parent.begin(), parent.end(), 0);
233 auto find = [&](int x) {
234 while (x != parent[x]) { parent[x] = parent[parent[x]]; x = parent[x]; }
235 return x;
236 };
237 for (const auto& e : edges_) {
238 int ra = find(e.from), rb = find(e.to);
239 if (ra != rb) parent[ra] = rb;
240 }
241 std::map<int, std::vector<int>> groups;
242 for (std::size_t i = 0; i < nodes_.size(); ++i)
243 groups[find(static_cast<int>(i))].push_back(static_cast<int>(i));
244
245 std::vector<std::vector<int>> result;
246 result.reserve(groups.size());
247 for (auto& [_, v] : groups) result.push_back(std::move(v));
248 return result;
249 }

251 // Breadth-first visit order starting from `start`. Honors graph kind.
252 std::vector<int> bfs(int start) const {
253 std::vector<int> order;
254 if (nodes_.empty()) return order;
255 std::vector<bool> seen(nodes_.size(), false);
256 std::queue<int> q;
257 q.push(start);
258 seen[start] = true;
259 while (!q.empty()) {
260 int x = q.front(); q.pop();
261 order.push_back(x);
262 for (int n : neighbors(x))
263 if (!seen[n]) { seen[n] = true; q.push(n); }
264 }
265 return order;
266 }
