// 1) Plain integer graph (undirected by default) auto g = Graph<int>::fromPairs({{0,1},{1,2},{2,0}});

// 2) With node deduplication struct EqInt { bool operator()(int a, int b) const { return a == b; } }; auto g = Graph<int, EqInt>::fromPairs({{1,2},{2,3}});

// 3) With weighted edges (e.g. Vector2f points + euclidean distance) struct EqV2 { bool operator()(Vector2f a, Vector2f b) const {...}; }; struct DistV2 { float operator()(Vector2f a, Vector2f b) const {...}; }; auto g = Graph<Vector2f, EqV2>::fromPairs<DistV2>(pairs); auto mst = g.mst(); // non-mutating auto loop = g.difference(mst); // remaining Delaunay edges → add back N for TinyKeep loops
